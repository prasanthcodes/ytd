Step 4: Once the scrolling action concludes, you can continue by pasting the second code snippet labeled as “Code snippet 2”. This specific code snippet is designed to extract the video titles and URLs from the YouTube playlist and showcase them in the console. Here is the code snippet for your convenience:


//COPY & PASTE CODE 2:

let seenUrls = new Map();
links.forEach(link => {
    let cleanHref = link.href.split('&list=')[0].split('&index=')[0];
    let title = link.getAttribute('title') || link.title || link.innerText;
    
    if (title && cleanHref.includes('watch')) {
        let cleanTitle = title.trim().replace(/[\r\n]+/g, " ");
        
        // Skip it entirely if the title is strictly a timestamp (e.g., 0:28, 14:22, 1:05:30)
        if (/^(\d{1,2}:)?\d{1,2}:\d{2}$/.test(cleanTitle)) {
            return; 
        }

        // If we haven't seen this URL yet, or if this new title is longer/better than the timestamp one
        if (!seenUrls.has(cleanHref) || cleanTitle.length > seenUrls.get(cleanHref).length) {
            seenUrls.set(cleanHref, cleanTitle);
        }
    }
});

// Convert our clean map back into your 'f' array format
arrayVideos = [];
seenUrls.forEach((title, url) => {
    // Clean any semicolons out of the title so it doesn't break your CSV columns
    let safeTitle = title.replace(/;/g, ',');
    arrayVideos.push(`${safeTitle};${url}`);
});
